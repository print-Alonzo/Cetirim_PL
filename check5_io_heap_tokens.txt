     4 | // The language has no pointer type, so there is nothing to dereference and
     5 | // no address to take. What it does have is a split allocation model, and
     6 | // that is what the second half of this program demonstrates: arrays and
     7 | // structs are allocated as heap objects and handled by reference, while
     8 | // scalars live in the call frame and are copied. Every aliasing effect
     9 | // below follows from that one rule.
    10 | 
    11 | struct Point {
    12 |     int x, y;
    13 | };
    14 | 
    15 | struct Node {
    16 |     string label;
    17 |     int weight;
    18 | };
    19 | 
    20 | // Scalar parameter: the callee writes to its own private copy.
    21 | void bump_scalar(int n) {
    22 |     n = n + 1000;
    23 | }
    24 | 
    25 | // Array parameter: caller and callee share one underlying array object.
    26 | void bump_array(int a[], int len) {
    27 |     var int i;
    28 |     for (i = 0; i < len; i = i + 1) {
    29 |         a[i] = a[i] + 1000;
    30 |     }
    31 | }
    32 | 
    33 | // Struct parameter: likewise shared, so a field write escapes the callee.
    34 | void bump_struct(struct Point p) {
    35 |     p.x = p.x + 1000;
    36 |     p.y = p.y + 1000;
    37 | }

    39 | // A scalar can only be "modified" by handing a new value back out.
    40 | int bumped(int n) {
